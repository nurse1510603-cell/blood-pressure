function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); try { var data = JSON.parse(e.postData.contents); var time = data.time || new Date().toLocaleString(); var user = data.user || '未知長輩'; var type = data.type || 'general'; var detailStr = ""; if (type === 'bp') { detailStr = `血壓紀錄 | 脈搏:${data.pulse} | 時段:${data.slot} | 狀態:${data.sys >= 130 || data.dia >= 80 ? '偏高' : '正常'}`; sheet.appendRow([time, user, data.sys, data.dia, data.pulse, data.slot, detailStr]); } else if (type === 'pain') { detailStr = `【疼痛】部位:${data.major}-${(data.subs||[]).join()} (${(data.sides||[]).join()}) | 型態:${(data.types||[]).join()} | 分數:${data.level}分 | 處置:${(data.reliefs||[]).join()} | 連結血壓:${data.linkedBP || '無'} | 備註:${data.note || ''}`; sheet.appendRow([time, user, 0, 0, 0, '疼痛紀錄', detailStr]); } else if (type === 'med') { detailStr = `【用藥設定】藥名:${data.name} | 時段:${(data.medTimes||[]).join()} | 置頂:${data.pinned ? '是':'否'}`; sheet.appendRow([time, user, 0, 0, 0, '用藥紀錄', detailStr]); } else if (type === 'cal') { detailStr = `【行事曆】類別:${data.cat} | 日期:${data.date} ${data.time} | 內容:${data.title}`; sheet.appendRow([time, user, 0, 0, 0, '行事曆行程', detailStr]); } else { sheet.appendRow([time, user, 0, 0, 0, type, JSON.stringify(data)]); } return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()})).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; result.push({ time: row[0], user: row[1], sys: row[2], dia: row[3], pulse: row[4], slot: row[5], note: row[6] }); } return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); }
